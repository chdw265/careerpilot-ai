    function careerEvidenceErrorMessage(error, fallback = "ApplyStronger could not save this Career Evidence entry.") {
      const raw = String(error?.message || error?.error_description || "").trim();
      const code = String(error?.code || "").trim();
      if (code === "40001" || /changed|reload|no longer valid/i.test(raw)) {
        return "This Career Evidence entry changed. The latest version will be reloaded before you continue.";
      }
      if (code === "P0002" || /not found/i.test(raw)) {
        return "This Career Evidence entry could not be found. Reload the page and try again.";
      }
      if (code === "42501" || /signed in|permission|not authorized/i.test(raw)) {
        return "Your session no longer has permission to make this change. Sign in again and retry.";
      }
      if (code === "55000" || /current state|immutable|already being analyzed/i.test(raw)) {
        return raw || "This Career Evidence entry cannot be changed in its current state.";
      }
      if (code === "22023" || /must|cannot|invalid|private identifier|characters/i.test(raw)) {
        return raw || fallback;
      }
      return raw || fallback;
    }

    function careerEvidenceNeedsReload(error) {
      const raw = String(error?.message || "");
      const code = String(error?.code || "");
      return code === "40001" || code === "P0002" || /changed|reload|no longer valid|not found/i.test(raw);
    }

    function approvedCareerEvidenceDestinationLabel(item) {
      if (item.destination === "career_profile") return "Career Profile skill";
      if (item.destination === "master_resume") return "Master resume bullet";
      return "Evidence Bank only";
    }

    function approvedCareerEvidenceCopyLabel(item) {
      if (item.destination === "evidence_bank") return "Verified evidence saved";
      if (item.destination_copy_active === true) return "Destination copy active";
      if (item.destination_copy_active === false) {
        const reason = String(item.destination_copy_deactivation_reason || "").replaceAll("_", " ");
        return reason ? `Destination copy inactive — ${reason}` : "Destination copy inactive";
      }
      return "Destination copy status unavailable";
    }

    function careerEvidenceSnippet(value, limit = 520) {
      const text = String(value || "").trim();
      return text.length > limit ? `${text.slice(0, limit - 3)}...` : text;
    }

    async function loadCareerEvidence(openEntryId = "") {
      if (!currentSession?.user) return;

      const [entriesResult, approvedItemsResult, experiencesResult] = await Promise.all([
        db
          .from("career_evidence_entries")
          .select("id,experience_id,context_label,original_text,current_text,status,analysis,reviewed_data,analysis_version,analysis_started_at,analyzed_at,reviewed_at,approved_at,archived_at,created_at,updated_at")
          .eq("user_id", currentSession.user.id)
          .order("updated_at", { ascending: false })
          .limit(50),
        db
          .from("career_evidence_approved_items")
          .select("id,evidence_id,analysis_version,item_type,source_suggestion,approved_text,source_excerpt,destination,profile_record_id,destination_copy_active,destination_copy_deactivated_at,destination_copy_deactivation_reason,approved_at")
          .eq("user_id", currentSession.user.id)
          .order("approved_at", { ascending: false })
          .limit(250),
        db
          .from("experiences")
          .select("id,job_title,company_name,start_date,end_date,is_current")
          .eq("user_id", currentSession.user.id)
          .order("start_date", { ascending: false, nullsFirst: false })
          .limit(100)
      ]);

      if (experiencesResult.error) {
        console.error("Career Evidence experience load error:", experiencesResult.error);
      }
      careerEvidenceExperienceMap = new Map((experiencesResult.data || []).map(item => [item.id, item]));
      addCareerEvidenceExperienceOptions(document.getElementById("careerEvidenceExperience"));

      if (entriesResult.error || approvedItemsResult.error) {
        console.error("Career Evidence load error:", entriesResult.error || approvedItemsResult.error);
        document.getElementById("careerEvidenceList").textContent = "Career Evidence is not available yet.";
        return;
      }

      const itemsByEvidence = new Map();
      (approvedItemsResult.data || []).forEach(item => {
        if (!itemsByEvidence.has(item.evidence_id)) itemsByEvidence.set(item.evidence_id, []);
        itemsByEvidence.get(item.evidence_id).push(item);
      });

      careerEvidenceEntries = (entriesResult.data || []).map(entry => ({
        ...entry,
        approved_items: itemsByEvidence.get(entry.id) || []
      }));
      renderCareerEvidenceList(openEntryId);
    }

    function reviewedCareerEvidenceItemMap(reviewed) {
      const result = new Map();
      const items = Array.isArray(reviewed?.items) ? reviewed.items : [];
      items.forEach(item => {
        const type = String(item?.item_type || "");
        const source = String(item?.source_suggestion || "");
        if (type && source) result.set(`${type}\u0000${source}`, item);
      });
      return result;
    }

    function collectCareerEvidenceReview(review) {
      const items = [...review.querySelectorAll("[data-evidence-item]")].map(wrapper => ({
        item_type: wrapper.dataset.itemType || "",
        source_suggestion: wrapper.dataset.sourceSuggestion || "",
        approved_text: wrapper.querySelector("[data-evidence-approved-text]")?.value.trim() || "",
        destination: wrapper.querySelector("[data-evidence-destination]")?.value || "",
        selected: Boolean(wrapper.querySelector("[data-evidence-select]")?.checked)
      }));
      return {
        experience_id: review.querySelector("[data-evidence-experience]")?.value || "",
        attested: Boolean(review.querySelector("[data-evidence-attestation]")?.checked),
        items
      };
    }

        summary.className = "career-evidence-card-copy";
        summary.textContent = analysis.summary;
        review.appendChild(summary);

      const relationshipFlags = [
        ...(Array.isArray(analysis.duplicate_candidates) ? analysis.duplicate_candidates.map(item => `Possible duplicate: ${item.reason || "Review similar saved evidence."}`) : []),
        ...(Array.isArray(analysis.possible_conflicts) ? analysis.possible_conflicts.map(item => `Possible conflict: ${item.reason || "Review potentially conflicting evidence."}`) : [])

        warning.className = "review-note";

        review.appendChild(warning);

      approvedItems.forEach(item => {

        row.className = "career-evidence-suggestion";
        const approvedText = document.createElement("div");

          support.textContent = `Approved evidence source: ${careerEvidenceSnippet(item.source_excerpt, 500)}`;
          row.appendChild(support);
        }
        card.appendChild(row);
      });
    }

    async function analyzeExistingCareerEvidence(entryId, button) {
      const original = button?.textContent || "Analyze Draft";
      if (button) {
        button.disabled = true;
        button.textContent = "Analyzing...";
      }
      try {
        const { data, error } = await db.functions.invoke("analyze-career-evidence", {
          body: { evidence_id: entryId }
        });
        if (error) throw new Error(await getFunctionErrorMessage(error, "ApplyStronger could not analyze this Career Evidence."));
        if (!data?.success) throw new Error(data?.error || "ApplyStronger could not analyze this Career Evidence.");
        await loadCareerEvidence(data.evidence?.id || entryId);
        showToast("Career Evidence is ready for your review.");
      } catch (error) {
        console.error("Career Evidence analysis error:", error);
        if (/changed|already being analyzed|not found/i.test(String(error?.message || ""))) await loadCareerEvidence(entryId);
        showToast(error?.message || "We couldn't analyze this Career Evidence.");
      } finally {
        if (button) {
          button.disabled = false;
          button.textContent = original;
        }
      }
    }

    async function createCareerEvidence() {
      if (!currentSession?.user) {
        openAuth("signin", "Sign in to build your Career Evidence.");
        return;
      }
      const textInput = document.getElementById("careerEvidenceText");
      const contextInput = document.getElementById("careerEvidenceContext");
      const experienceInput = document.getElementById("careerEvidenceExperience");
      const button = document.getElementById("analyzeCareerEvidenceButton");
      const status = document.getElementById("careerEvidenceStatus");
      const originalText = textInput.value.trim();
      if (originalText.length < 20) {
        status.textContent = "Describe at least one responsibility, project, tool, or accomplishment.";
        textInput.focus();
        return;
      }
      if (/\b\d{3}-\d{2}-\d{4}\b/.test(originalText) || /\b(?:medical record number|mrn)\s*[:#-]?\s*[a-z0-9-]{4,}\b/i.test(originalText)) {
        status.textContent = "Remove private identifiers before continuing.";
        return;
      }

      button.disabled = true;
      status.textContent = "Saving your evidence...";
      let entryId = "";
      try {
        const { data: created, error: createError } = await db.rpc("applystronger_create_career_evidence", {
          p_original_text: originalText,
          p_context_label: contextInput.value.trim() || null,
          p_experience_id: experienceInput.value || null
        });
        if (createError) throw createError;
        entryId = created?.evidence?.id || "";
        if (!created?.success || !entryId) throw new Error("Career Evidence could not be saved.");

        status.textContent = "Analyzing your evidence...";
        const { data, error } = await db.functions.invoke("analyze-career-evidence", {
          body: { evidence_id: entryId }
        });
        if (error) throw new Error(await getFunctionErrorMessage(error, "ApplyStronger could not analyze this Career Evidence."));
        if (!data?.success) throw new Error(data?.error || "ApplyStronger could not analyze this Career Evidence.");

        textInput.value = "";
        contextInput.value = "";
        experienceInput.value = "";
        status.textContent = "Analysis ready. Review every suggestion before approval.";
        await loadCareerEvidence(data.evidence?.id || entryId);
        showToast("Career Evidence is ready for your review.");
      } catch (error) {
        console.error("Create Career Evidence error:", error);
        status.textContent = error instanceof Error && entryId
          ? `${error.message} Your draft was saved and can be analyzed again.`
          : careerEvidenceErrorMessage(error, "We couldn't save this Career Evidence.");
        await loadCareerEvidence(entryId);
      } finally {
        button.disabled = false;
      }
    }

    async function saveCareerEvidenceDraft(entry, editor, button) {
      const status = editor.querySelector("[data-evidence-edit-status]");
      const currentText = editor.querySelector("[data-evidence-edit-text]")?.value.trim() || "";
      const contextLabel = editor.querySelector("[data-evidence-edit-context]")?.value.trim() || "";
      const experienceId = editor.querySelector("[data-evidence-edit-experience]")?.value || "";
      if (currentText.length < 20) {
        status.textContent = "Describe at least one responsibility, project, tool, or accomplishment.";
        return;
      }

      button.disabled = true;
      status.textContent = "Saving working statement...";
      try {
        const { data, error } = await db.rpc("applystronger_update_career_evidence", {
          p_evidence_id: entry.id,
          p_current_text: currentText,
          p_context_label: contextLabel || null,
          p_experience_id: experienceId || null,
          p_expected_updated_at: entry.updated_at
        });
        if (error) throw error;
        if (!data?.success) throw new Error("Career Evidence could not be updated.");
        invalidateCareerEvidenceDependentState();
        await loadCareerEvidence(entry.id);
        showToast("Working statement saved. Analyze it again before approval.");
      } catch (error) {
        console.error("Career Evidence update error:", error);
        status.textContent = careerEvidenceErrorMessage(error, "We couldn't update this Career Evidence.");
        if (careerEvidenceNeedsReload(error)) await loadCareerEvidence(entry.id);
      } finally {
        button.disabled = false;
      }
    }

    async function saveCareerEvidenceReview(entry, review, approve, button) {
      const payload = collectCareerEvidenceReview(review);
      const status = review.querySelector("[data-evidence-review-status]");
      const selected = payload.items.filter(item => item.selected);
      if (approve && !selected.length) {
        status.textContent = "Select at least one truthful item before approval.";
        return;
      }
      if (selected.some(item => !item.destination)) {
        status.textContent = "Choose a destination for every selected item.";
        return;
      }
      if (selected.some(item => item.item_type === "resume_bullet" && item.destination === "master_resume") && !payload.experience_id) {
        status.textContent = "Choose the work experience that should receive selected master-resume bullets.";
        return;
      }
      if (approve && !payload.attested) {
        status.textContent = "Confirm the accuracy statement before approving Career Evidence.";
        return;
      }

      button.disabled = true;
      status.textContent = approve ? "Approving selected items..." : "Saving your review...";
      try {
        const rpcName = approve
          ? "applystronger_approve_career_evidence"
          : "applystronger_save_career_evidence_review";
        const { data, error } = await db.rpc(rpcName, {
          p_evidence_id: entry.id,
          p_reviewed_data: payload,
          p_expected_updated_at: entry.updated_at
        });
        if (error) throw error;
        if (!data?.success) throw new Error(approve
          ? "ApplyStronger could not approve these Career Evidence selections."
          : "ApplyStronger could not save this Career Evidence review.");

        if (approve) {
          invalidateCareerEvidenceDependentState();
          await Promise.all([
            loadCareerEvidence(),
            loadCareerProfile(currentSession.user.id),
            loadMemberDashboard(currentSession.user.id)
          ]);
          showToast("Career Evidence approved. Only the destinations you selected were updated.");
        } else {
          await loadCareerEvidence(entry.id);
          showToast("Career Evidence review saved. Nothing was added to your Career Profile or resume.");
        }
      } catch (error) {
        console.error("Career Evidence review error:", error);
        status.textContent = careerEvidenceErrorMessage(error, "We couldn't save this Career Evidence review.");
        if (careerEvidenceNeedsReload(error)) await loadCareerEvidence(entry.id);
      } finally {
        button.disabled = false;
      }
    }

    async function archiveCareerEvidence(entry, button) {
      if (entry.status === "approved") {
        const confirmed = window.confirm("Archive this Career Evidence? It will stop supporting future Application Readiness reviews. Skills or resume bullets you already approved into your profile will remain independently editable.");
        if (!confirmed) return;
      }
      button.disabled = true;
      const original = button.textContent;
      button.textContent = "Archiving...";
      try {
        const { data, error } = await db.rpc("applystronger_archive_career_evidence", {
          p_evidence_id: entry.id,
          p_expected_updated_at: entry.updated_at
        });
        if (error) throw error;
        if (!data?.success) throw new Error("ApplyStronger could not archive this Career Evidence.");
        invalidateCareerEvidenceDependentState();
        await loadCareerEvidence();
        showToast("Career Evidence archived. Its audit history was preserved.");
      } catch (error) {
        console.error("Career Evidence archive error:", error);
        showToast(careerEvidenceErrorMessage(error, "We couldn't archive this Career Evidence."));
        if (careerEvidenceNeedsReload(error)) await loadCareerEvidence(entry.id);
      } finally {
        button.disabled = false;
        button.textContent = original;
      }
    }
